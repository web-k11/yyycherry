/** * 主入口：網頁讀取時觸發 */ function doGet(e) { var cache = CacheService.getScriptCache(); // 👉 2. 先看總共切了幾片 var chunkCountStr = cache.get("final_payload_chunks"); if (chunkCountStr != null) { var chunkCount = parseInt(chunkCountStr, 10); var fullData = ""; var isComplete = true; // 👉 2. 把碎片拼起來 for (var i = 0; i < chunkCount; i++) { var chunk = cache.get("final_payload_p" + i); if (chunk == null) { isComplete = false; break; } fullData += chunk; } if (isComplete) { return ContentService.createTextOutput(fullData).setMimeType(ContentService.MimeType.JSON); } } // 如果快取空了，重新產生 var data = refreshDataCache(); return ContentService.createTextOutput(data).setMimeType(ContentService.MimeType.JSON); } /** * 預處理邏輯：讀取、過濾、並存入快取 */ function refreshDataCache() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var result = { titleData: [], sheet1Data: [] }; // 1. 處理 title var titleSheet = ss.getSheetByName("title"); if (titleSheet) { var tData = titleSheet.getDataRange().getValues(); var tHeaders = tData[0]; for (var i = 1; i < tData.length; i++) { var obj = {}; for (var j = 0; j < tHeaders.length; j++) { var h = String(tHeaders[j]).trim(); if (h !== "") obj[h] = tData[i][j]; } result.titleData.push(obj); } } // 2. 處理 Sheet1 var sheet1 = ss.getSheetByName("Sheet1"); if (sheet1) { var sData = sheet1.getDataRange().getValues(); var sHeaders = sData[0]; var processedHeaders = sHeaders.map(function(h) { var hStr = String(h).trim(); return hStr.length > 10 ? "" : hStr; }); for (var x = 1; x < sData.length; x++) { var sObj = {}; for (var y = 0; y < processedHeaders.length; y++) { var curHeader = processedHeaders[y]; if (curHeader !== "") sObj[curHeader] = sData[x][y]; } result.sheet1Data.push(sObj); } } var finalJson = JSON.stringify(result); var cache = CacheService.getScriptCache(); // 👉 2. 自動切片寫入快取 (每片約 90KB) var chunkSize = 90000; var chunks = Math.ceil(finalJson.length / chunkSize); // 清除可能殘留的舊切片 cache.remove("final_payload_chunks"); for (var c = 0; c < 20; c++) { cache.remove("final_payload_p" + c); } try { for (var n = 0; n < chunks; n++) { var segment = finalJson.substring(n * chunkSize, (n + 1) * chunkSize); cache.put("final_payload_p" + n, segment, 21600); } cache.put("final_payload_chunks", chunks.toString(), 21600); } catch (err) { console.error("快取失敗: " + err); } return finalJson; } /** * 自動觸發：當試算表內容變動時 */ function onEditTrigger(e) { var cache = CacheService.getScriptCache(); // 👉 4. 只負責刪除快取，讓下一次請求自動重新抓資料 cache.remove("final_payload_chunks"); for (var i = 0; i < 20; i++) { cache.remove("final_payload_p" + i); } }
